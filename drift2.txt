import pandas as pd
import numpy as np

def numerical_distance(df1, col1, df2, col2, debug=False):
    """
    Exact Python equivalent of Scala numericalDistance().
    Computes max absolute CDF difference between two datasets.
    
    Parameters:
        df1, col1 : DataFrame and numeric column for dataset 1 (data_x)
        df2, col2 : DataFrame and numeric column for dataset 2 (data_y)
        debug     : If True, prints the full CDF table
    
    Returns:
        max_diff (float), at_value (float)
    """
    
    # 1️⃣ Extract numeric values & drop NaNs
    vals1 = pd.to_numeric(df1[col1], errors='coerce').dropna().astype(float).values
    vals2 = pd.to_numeric(df2[col2], errors='coerce').dropna().astype(float).values
    
    n = len(vals1)
    m = len(vals2)
    if n == 0 or m == 0:
        raise ValueError("One of the datasets has no valid numeric values.")
    
    # 2️⃣ Build rank maps { value: rank }
    def build_rank_map(arr):
        sorted_vals = np.sort(arr)
        rank_map = {}
        for val in np.unique(sorted_vals):
            rank_map[val] = np.searchsorted(sorted_vals, val, side='right')  # count ≤ val
        return rank_map
    
    rankMap1 = build_rank_map(vals1)
    rankMap2 = build_rank_map(vals2)
    
    # 3️⃣ Union of all unique values
    combined_keys = sorted(set(rankMap1.keys()) | set(rankMap2.keys()))
    
    # 4️⃣ Loop over keys and calculate max diff
    linfSimple = 0.0
    table = []
    
    for key in combined_keys:
        cdf1 = rankMap1.get(key, max((r for v, r in rankMap1.items() if v < key), default=0)) / n
        cdf2 = rankMap2.get(key, max((r for v, r in rankMap2.items() if v < key), default=0)) / m
        diff = abs(cdf1 - cdf2)
        table.append((key, cdf1, cdf2, diff))
        linfSimple = max(linfSimple, diff)
    
    # 5️⃣ Debug table option
    if debug:
        print(f"{'Value':>12} {'CDF1':>10} {'CDF2':>10} {'AbsDiff':>10}")
        for v, c1, c2, d in table:
            print(f"{v:>12.2f} {c1:>10.4f} {c2:>10.4f} {d:>10.4f}")
    
    # 6️⃣ Get value where max diff occurs
    max_row = max(table, key=lambda x: x[3])
    return linfSimple, max_row[0]
