proc sql;
  create table odm_c5 as
  select a.cntr_mbr_id, 
         a.mo_end_dt,
         a.ldp as def1,
         b.ldp as def2,
         c.ldp as def3,
         case 
           when a.ldp <= b.ldp and 
                b.ldp <= c.ldp and 
                a.ldp < c.ldp 
                then 1
           when a.ldp >= b.ldp and 
                b.ldp >= c.ldp and 
                a.ldp > c.ldp 
                then -1
           else .
         end as target
  from odm_c4 as a
    left join odm_c4 as b 
      on a.cntr_mbr_id = b.cntr_mbr_id and b.mo_end_dt = a.mo_end_dt + 1
    left join odm_c4 as c 
      on a.cntr_mbr_id = c.cntr_mbr_id and c.mo_end_dt = a.mo_end_dt + 2
  order by a.cntr_mbr_id, a.mo_end_dt;
quit;
