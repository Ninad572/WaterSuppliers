import numpy as np import pandas as pd import matplotlib.pyplot as plt

SEED = 42 np.random.seed(SEED)

def create_synthetic_dataset(num_rows=100): data = { "participant_id": np.arange(num_rows), "income": np.maximum(1000, np.random.normal(loc=60000, scale=15000, size=num_rows)).astype(int), "gender": np.random.choice(["Male", "Female"], size=num_rows), "region": np.random.choice(["North", "South", "East", "West"], size=num_rows), } df = pd.DataFrame(data) def true_ctr_arm_0(row): return np.clip(0.1 + (row["income"] / 150000) * 0.4 + (0.2 if row["gender"] == "Male" and row["region"] == "North" else 0.0), 0.0, 1.0) def true_ctr_arm_1(row): return np.clip(0.1 + (row["income"] / 200000) * 0.3 + (0.3 if row["gender"] == "Female" and row["region"] == "South" else 0.0), 0.0, 1.0) def true_ctr_arm_2(row): return np.clip(0.1 + (row["income"] / 100000) * 0.5 + (0.1 if row["region"] == "West" else 0.0), 0.0, 1.0) df["true_ctr_arm_0"] = df.apply(true_ctr_arm_0, axis=1) df["true_ctr_arm_1"] = df.apply(true_ctr_arm_1, axis=1) df["true_ctr_arm_2"] = df.apply(true_ctr_arm_2, axis=1) return df

def preprocess_data(df): df_encoded = pd.get_dummies(df, columns=["gender", "region"], dtype=float) df_encoded["income"] = (df_encoded["income"] - df_encoded["income"].min()) / (df_encoded["income"].max() - df_encoded["income"].min()) context_features = df_encoded.drop(columns=["participant_id", "true_ctr_arm_0", "true_ctr_arm_1", "true_ctr_arm_2"]) true_ctrs = df[["true_ctr_arm_0", "true_ctr_arm_1", "true_ctr_arm_2"]] return context_features, true_ctrs

def run_ab_testing(true_ctrs, num_iterations=100): num_arms = true_ctrs.shape[1] rewards = [] chosen_arms = [] for i in range(num_iterations): chosen_arm = i % num_arms ctr = true_ctrs.iloc[i % len(true_ctrs), chosen_arm] reward = np.random.binomial(1, ctr) rewards.append(reward) chosen_arms.append(chosen_arm) return np.array(rewards), np.array(chosen_arms)

def run_random(true_ctrs, num_iterations=100): num_arms = true_ctrs.shape[1] rewards = [] for i in range(num_iterations): chosen_arm = np.random.randint(num_arms) ctr = true_ctrs.iloc[i % len(true_ctrs), chosen_arm] rewards.append(np.random.binomial(1, ctr)) return np.array(rewards)

def run_oracle(true_ctrs, num_iterations=100): rewards = [] for i in range(num_iterations): ctrs = true_ctrs.iloc[i % len(true_ctrs)] best_ctr = np.max(ctrs) rewards.append(np.random.binomial(1, best_ctr)) return np.array(rewards)

if name == "main": num_rows = 500 num_iterations = 500 raw_df = create_synthetic_dataset(num_rows=num_rows) _, true_ctrs = preprocess_data(raw_df) ab_rewards, ab_chosen_arms = run_ab_testing(true_ctrs, num_iterations) random_rewards = run_random(true_ctrs, num_iterations) oracle_rewards = run_oracle(true_ctrs, num_iterations) plt.figure(figsize=(12, 4)) plt.plot(np.cumsum(ab_rewards), label="A/B Testing") plt.plot(np.cumsum(oracle_rewards), label="Oracle") plt.plot(np.cumsum(random_rewards), label="Random") plt.xlabel("Iteration") plt.ylabel("Cumulative Reward") plt.legend() plt.title("Cumulative Reward Comparison") plt.show() oracle_cumsum = np.cumsum(oracle_rewards) ab_cumsum = np.cumsum(ab_rewards) regret = oracle_cumsum - ab_cumsum plt.figure(figsize=(12, 4)) plt.plot(regret, label="Regret (Oracle - A/B)") plt.xlabel("Iteration") plt.ylabel("Regret") plt.legend() plt.title("Regret Over Time") plt.show() plt.figure(figsize=(12, 4)) plt.hist(ab_chosen_arms, bins=np.arange(4)-0.5, rwidth=0.8) plt.xticks(range(3)) plt.xlabel("Arm") plt.ylabel("Frequency") plt.title("Arm Selection Distribution (A/B Testing)") plt.show() print(f"A/B Testing Total Reward: {ab_rewards.sum()}") print(f"Random Policy Total Reward: {random_rewards.sum()}") print(f"Oracle Total Reward: {oracle_rewards.sum()}")
