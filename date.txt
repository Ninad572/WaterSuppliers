# ============================
# ORDINAL REGRESSION USING XGBOOST (70-30 SPLIT)
# ============================

import numpy as np
import pandas as pd

from xgboost import XGBRegressor
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_absolute_error, mean_squared_error, cohen_kappa_score

# ----------------------------
# 1. INPUT DATA
# ----------------------------
# Assumptions:
# - df is your dataframe
# - TARGET column contains ordinal values (a–h)
# - Feature preprocessing already done

TARGET_COL = "TARGET"

X = df.drop(columns=[TARGET_COL])
y = df[TARGET_COL]

# ----------------------------
# 2. ORDINAL ENCODING
# ----------------------------
ordinal_levels = sorted(y.unique())
ordinal_map = {label: idx for idx, label in enumerate(ordinal_levels)}
reverse_ordinal_map = {v: k for k, v in ordinal_map.items()}

y_encoded = y.map(ordinal_map)

# ----------------------------
# 3. TRAIN / VALIDATION SPLIT (70 / 30)
# ----------------------------
X_train, X_val, y_train, y_val = train_test_split(
    X,
    y_encoded,
    test_size=0.30,     # 30% validation
    random_state=42
)

# ----------------------------
# 4. XGBOOST REGRESSOR
# ----------------------------
model = XGBRegressor(
    objective="reg:squarederror",
    n_estimators=500,
    max_depth=6,
    learning_rate=0.05,
    subsample=0.8,
    colsample_bytree=0.8,
    reg_alpha=0.0,
    reg_lambda=1.0,
    random_state=42,
    n_jobs=-1
)

model.fit(
    X_train,
    y_train,
    eval_set=[(X_val, y_val)],
    verbose=False
)

# ----------------------------
# 5. PREDICTION
# ----------------------------
y_val_pred_cont = model.predict(X_val)

# Clamp predictions to valid ordinal range
y_val_pred_cont = np.clip(
    y_val_pred_cont,
    y_encoded.min(),
    y_encoded.max()
)

# Round to nearest ordinal level
y_val_pred_ord = np.rint(y_val_pred_cont).astype(int)

# ----------------------------
# 6. EVALUATION METRICS
# ----------------------------
mae = mean_absolute_error(y_val, y_val_pred_cont)
rmse = np.sqrt(mean_squared_error(y_val, y_val_pred_cont))

qwk = cohen_kappa_score(
    y_val,
    y_val_pred_ord,
    weights="quadratic"
)

print("========== MODEL EVALUATION ==========")
print(f"MAE  : {mae:.4f}")
print(f"RMSE : {rmse:.4f}")
print(f"QWK  : {qwk:.4f}")

# ----------------------------
# 7. FINAL ORDINAL PREDICTIONS
# ----------------------------
y_val_pred_labels = pd.Series(y_val_pred_ord).map(reverse_ordinal_map)

val_results = X_val.copy()
val_results["ACTUAL"] = y_val.map(reverse_ordinal_map).values
val_results["PREDICTED"] = y_val_pred_labels.values
