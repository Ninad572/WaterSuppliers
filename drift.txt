import numpy as np

def max_cdf_difference_numpy(vals1, vals2):
    """
    Compute max CDF difference (KS statistic) between two numeric arrays using NumPy.
    vals1: 1D array-like of floats/ints (Dataset 1, vertical 12 months per ID)
    vals2: 1D array-like of floats/ints (Dataset 2, 1 per ID)
    Returns (max_diff, at_value)
    """
    # Convert to numpy, drop NaNs
    x = np.asarray(vals1, dtype=float)
    y = np.asarray(vals2, dtype=float)
    x = x[~np.isnan(x)]
    y = y[~np.isnan(y)]
    n = x.size
    m = y.size
    if n == 0 or m == 0:
        raise ValueError("One of the datasets has no valid data.")

    # Sort
    x.sort()
    y.sort()

    # Unique values and counts (to construct step CDFs efficiently)
    ux, cx = np.unique(x, return_counts=True)
    uy, cy = np.unique(y, return_counts=True)

    # Cumulative counts -> CDF steps at unique points
    cfx = np.cumsum(cx) / n
    cfy = np.cumsum(cy) / m

    # Merge the two step functions over the union of unique values
    u = np.union1d(ux, uy)

    # For each u, find rightmost index in ux, uy where value <= u,
    # then read the CDF value from cfx/cfy. If no such index, CDF=0.
    idx_x = np.searchsorted(ux, u, side='right') - 1
    idx_y = np.searchsorted(uy, u, side='right') - 1

    # Gather CDF values with boundary handling
    cdf_x = np.where(idx_x >= 0, cfx[idx_x], 0.0)
    cdf_y = np.where(idx_y >= 0, cfy[idx_y], 0.0)

    diff = np.abs(cdf_x - cdf_y)
    max_idx = np.argmax(diff)
    return float(diff[max_idx]), float(u[max_idx])
