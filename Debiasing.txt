import pandas as pd

# Assuming your dataset is named data_s and has columns: PRTCP_ID, YEAR, MONTH, DFRL_PCT, PRIOR_MONTH_DFRL_PCT

# Ensure correct sorting for each ID over time
data_s = data_s.sort_values(by=['PRTCP_ID', 'YEAR', 'MONTH']).reset_index(drop=True)

# Function to determine target for each row
def get_target(df):
    df['TARGET'] = 0  # default target

    for i in range(len(df)):
        current = df.loc[i, 'DFRL_PCT']
        prior = df.loc[i, 'PRIOR_MONTH_DFRL_PCT']

        if pd.isna(current) or pd.isna(prior):
            continue

        # Check increase and non-decreasing next 3 months
        if current > prior and i + 3 < len(df) and df.loc[i:i+3, 'DFRL_PCT'].is_monotonic_increasing:
            df.loc[i, 'TARGET'] = 1
        # Check decrease and non-increasing next 3 months
        elif current < prior and i + 3 < len(df) and df.loc[i:i+3, 'DFRL_PCT'].is_monotonic_decreasing:
            df.loc[i, 'TARGET'] = -1

    return df

# Apply the logic group-wise
data_s = data_s.groupby('PRTCP_ID', group_keys=False).apply(get_target)
