def common_decile_edges(series):
    """
    Computes common decile edges from the baseline series.
    Includes (-inf, ...) and (..., inf) overflow edges.
    Handles NaNs and ties.
    """
    clean_vals = pd.to_numeric(series, errors='coerce').dropna().astype(float)
    if clean_vals.empty:
        raise ValueError("Baseline series has no valid numeric values.")
    # Deciles at 10%, 20%, ..., 90%
    qs = np.quantile(clean_vals, q=[i/10 for i in range(1, 10)], method='linear')
    qs = np.unique(qs)  # Remove duplicate cut points
    edges = np.concatenate(([-np.inf], qs, [np.inf]))
    return edges

def apply_edges_to_series(series, edges):
    """
    Assigns bin numbers 1..K to the series based on edges.
    """
    vals = pd.to_numeric(series, errors='coerce').astype(float)
    bins = np.digitize(vals, edges[1:-1], right=True) + 1
    return bins
