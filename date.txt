proc sql;
    /* Create a temporary table with row numbers by asset group */
    create table temp as 
    select *, 
           monotonic() as row_num 
    from (select * from your_table order by asset_group, days);

    /* Count total rows per asset group */
    create table counts as 
    select asset_group, count(*) as total_rows 
    from your_table 
    group by asset_group;
quit;

data temp2;
    merge temp counts;
    by asset_group;
    lower_bound = ceil(total_rows * 0.1);
    upper_bound = floor(total_rows * 0.9);
run;

proc sql;
    /* Calculate the middle 80% average per asset group */
    select asset_group, mean(days) as middle_80_avg
    from temp2
    where row_num > lower_bound and row_num <= upper_bound
    group by asset_group;
quit;
