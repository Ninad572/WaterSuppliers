import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# -------------------------------
# Sample DataFrame creation
# -------------------------------
# Replace this with your actual data
np.random.seed(0)
data1 = pd.DataFrame({
    'eid_pid': range(1, 101),
    'num1': np.random.normal(50, 10, size=100),
    'num2': np.random.uniform(0, 100, size=100),
    'target': np.random.choice([-1, 0, 1], size=100)
})

# List of numerical variables
num_vars = ['num1', 'num2']
target_col = 'target'

# -------------------------------
# Plotting for each numerical variable
# -------------------------------
for col in num_vars:
    # Create 10 equal-width bins for the numerical variable
    data1[f'{col}_bin'] = pd.cut(data1[col], bins=10)

    # Count total per bin
    total_counts = data1.groupby(f'{col}_bin')['eid_pid'].count().rename('total')

    # Count target distribution per bin
    target_counts = data1.groupby([f'{col}_bin', target_col])['eid_pid'].count().unstack(fill_value=0)

    # Combine into one DataFrame
    aggregated = total_counts.to_frame().join(target_counts)
    aggregated = aggregated.reset_index()

    # Convert bin labels to strings for plotting
    bin_labels = aggregated[f'{col}_bin'].astype(str)

    # Plot
    fig, ax1 = plt.subplots(figsize=(12, 6))
    ax2 = ax1.twinx()

    # Bar plot of total count
    ax1.bar(bin_labels, aggregated['total'], color='lightgray', alpha=0.7, label='Total Count')

    # Line plots for proportions
    colors = {-1: 'red', 0: 'green', 1: 'blue'}
    for target_value, color in colors.items():
        if target_value in aggregated.columns:
            proportion = aggregated[target_value] / aggregated['total']
            ax2.plot(bin_labels, proportion, marker='o', color=color, label=f"Proportion Target {target_value}")

    ax1.set_xlabel(f'{col} (binned)')
    ax1.set_ylabel('Total Count')
    ax2.set_ylabel('Proportion of Target')
    ax1.set_title(f"Distribution for {col} (Binned)")
    ax1.tick_params(axis='x', rotation=45)

    # Combine legends
    lines1, labels1 = ax1.get_legend_handles_labels()
    lines2, labels2 = ax2.get_legend_handles_labels()
    ax1.legend(lines1 + lines2, labels1 + labels2, loc='upper right')

    plt.tight_layout()
    plt.show()

    # Clean up the bin column
    data1.drop(columns=[f'{col}_bin'], inplace=True)
